import { test, expect } from "vitest";

/** Turn arbitrary text into a URL-safe slug. */
const slugify = (input: string, maxLength = 64) =>
  input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength);

/** Output only ever contains lowercase letters, digits and dashes. */
test("slugify > Charset", () => {
  const actual = slugify("  Ünïcode & symbols ~~ 42 ");
  const expected = "^[a-z0-9-]*$";
  expect(actual).toMatch(new RegExp("^[a-z0-9-]*$"));
});
