import { test, expect } from "vitest";

function truncate(input: string, max: number, ellipsis = "…"): string {
  if (input.length <= max) return input;
  return input.slice(0, Math.max(0, max - ellipsis.length)) + ellipsis;
}

test("Tests > truncate > CustomEllipsis", async () => {
  expect(await truncate("hello world", 8, "...")).toSatisfy((s: string) => s.endsWith("..."));
});
