import { test, expect } from "vitest";
import { readFileSync } from "node:fs";

interface User {
  id: number;
  name: string;
  email: string;
  roles: string[];
  createdAt: string;
}

test("UserRepository > SeedShape", () => {
  const Seed = JSON.parse(readFileSync(new URL("./fixtures/users.json", import.meta.url), "utf8"));
  const actual1 = Seed.length;
  const expected1 = 0;
  expect.soft(actual1).toBeGreaterThan(expected1);
  const actual2 = Seed[0]!.email;
  const expected2 = "@";
  expect.soft(actual2).toContain(expected2);
});
