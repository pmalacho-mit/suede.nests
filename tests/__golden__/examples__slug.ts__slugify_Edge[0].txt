import { test, expect } from "vitest";

/** Turn arbitrary text into a URL-safe slug. */
const slugify = (input: string, maxLength = 64) =>
  input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength);

test("slugify > Edge[0]", () => {
  const actual = slugify("");
  const expected = "";
  expect(actual).toEqual(expected);
});
