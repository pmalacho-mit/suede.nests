import { test, expect } from "vitest";

/** Turn arbitrary text into a URL-safe slug. */
function slugify(input: string, maxLength = 64): string {
  return input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength);
}

test("Tests > slugify > NoLeadingOrTrailingDash", () => {
  const actual1 = slugify("--foo--");
  const expected1 = "f";
  expect.soft(actual1).toSatisfy((s: string) => s.startsWith(expected1));
  const actual2 = slugify("--foo--");
  const expected2 = "o";
  expect.soft(actual2).toSatisfy((s: string) => s.endsWith(expected2));
  const actual3 = slugify("--foo--");
  const expected3 = "--";
  expect.soft(actual3).not.toContain(expected3);
});
