import { test, expect } from "vitest";

interface User {
  id: number;
  name: string;
  email: string;
  roles: string[];
  createdAt: string;
}

interface Clock {
  now(): string;
}

class UserRepository {
  private users = new Map<number, User>();
  private nextId = 1;
  private readonly clock: Clock;

  constructor(clock: Clock) {
    this.clock = clock;
  }

  async create(
    input: Pick<User, "name" | "email"> & Partial<Pick<User, "roles">>,
  ): Promise<User> {
    if (!input.email.includes("@"))
      throw new TypeError(`invalid email: ${input.email}`);
    const user: User = {
      id: this.nextId++,
      roles: [],
      ...input,
      createdAt: this.clock.now(),
    };
    this.users.set(user.id, user);
    return user;
  }

  async find(id: number): Promise<User | undefined> {
    return this.users.get(id);
  }

  async list(): Promise<User[]> {
    return [...this.users.values()];
  }

  async grant(id: number, role: string): Promise<void> {
    const user = this.users.get(id);
    if (!user) throw new Error(`no user ${id}`);
    if (!user.roles.includes(role)) user.roles.push(role);
  }
}

const isAdmin = (u: User) => u.roles.includes("admin");
const hasEmailDomain = (domain: string) => (u: User) =>
  u.email.endsWith(`@${domain}`);

// Test-only helper. Being in the module is fine: it is tiny and tree-shakes away when unused.
const fixedNow = () => "2026-01-01T00:00:00.000Z";

/** grant() is idempotent and visible through every handle to the user. */
test("Tests > UserRepository > Grant", async () => {
  const FixedClock: Clock = { now: fixedNow };
  const Repo = new UserRepository(FixedClock);
  const Ada = await Repo.create({ name: "Ada", email: "ada@example.com" });
  await Repo.grant(Ada.id, "admin");
  await Repo.grant(Ada.id, "admin");
  const actual1 = Ada.roles;
  const expected1 = ["admin"];
  expect.soft(actual1).toEqual(expected1);
  expect.soft(Ada).toSatisfy(isAdmin);
  const actual3 = await Repo.list();
  expect.soft(actual3).toSatisfy((xs: ArrayLike<any>) => Array.from(xs).some(isAdmin));
  const actual4 = await Repo.list();
  const expected4 = hasEmailDomain("example.com");
  expect.soft(actual4).toSatisfy((xs: ArrayLike<any>) => Array.from(xs).every(expected4));
});
