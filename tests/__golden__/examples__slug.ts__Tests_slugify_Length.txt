import { test, expect } from "vitest";

/** Turn arbitrary text into a URL-safe slug. */
function slugify(input: string, maxLength = 64): string {
  return input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength);
}

/** `maxLength` is a hard cap. */
test("Tests > slugify > Length", () => {
  const actual = slugify("a very long title that goes on and on and on and on and on and on", 16).length;
  const expected = 16;
  expect(actual).toBeLessThanOrEqual(expected);
});
