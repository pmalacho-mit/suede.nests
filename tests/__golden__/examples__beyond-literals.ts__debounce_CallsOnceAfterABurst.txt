import { test, expect, vi, onTestFinished } from "vitest";

const debounce = (run: () => void, ms: number) => {
  let timer: ReturnType<typeof setTimeout> | undefined;
  return () => {
    clearTimeout(timer);
    timer = setTimeout(run, ms);
  };
};

/** a burst of calls runs once, after the quiet period */
test("debounce > CallsOnceAfterABurst", async () => {
  vi.useFakeTimers();
  onTestFinished(() => { vi.useRealTimers(); });
  const Run = vi.fn();
  const Debounced = debounce(Run, 100);
  Debounced();
  Debounced();
  Debounced();
  await vi.advanceTimersByTimeAsync(100);
  const actual = Run.mock.calls.length;
  const expected = 1;
  expect(actual).toEqual(expected);
});
