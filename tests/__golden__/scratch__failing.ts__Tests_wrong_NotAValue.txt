import { test, expect } from "vitest";
const nt_unsupported = (t: string) => { throw new Error(`namespace-tests: cannot materialize \`${t}\``); };

const add = (a: number, b: number, c = 0) => a + b + c;

test("Tests > wrong > NotAValue", () => {
  const actual = add(1_000_000, 2_000_000, 30);
  const expected = nt_unsupported("number");
  expect(actual).toEqual(expected);
});
