import { test, expect } from "vitest";

// ─── A reducer ─────────────────────────────────────────────────────────────

interface Todo {
  id: bigint;
  text: string;
  done: boolean;
  due: string | null;
}
interface State {
  todos: Todo[];
  filter: "all" | "open" | "done";
}

const initialState: State = { todos: [], filter: "all" };

/** The exported `initialState` singleton is usable directly via `typeof`. */
test("Tests > reduce > InitialIsEmpty", async () => {
  expect((initialState).todos).toHaveLength(0);
});
