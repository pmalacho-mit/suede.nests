import { test, expect } from "vitest";

/** Turn arbitrary text into a URL-safe slug. */
function slugify(input: string, maxLength = 64): string {
  return input
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, maxLength);
}

/** Output only ever contains lowercase letters, digits and dashes. */
test("Tests > slugify > Charset", async () => {
  expect(await slugify("  Ünïcode & symbols ~~ 42 ")).toMatch(new RegExp("^[a-z0-9-]*$"));
});
