import { test, expect } from "vitest";
import { existsSync, readFileSync } from "node:fs";

const countUsers = (users: unknown[]) => users.length;

/** the fixture is in this repository, so the test runs */
test.skipIf(!existsSync(new URL("./fixtures/users.json", import.meta.url)))("countUsers > Shipped", () => {
  const actual = countUsers(JSON.parse(readFileSync(new URL("./fixtures/users.json", import.meta.url), "utf8")));
  const expected = 0;
  expect(actual).toBeGreaterThan(expected);
});
