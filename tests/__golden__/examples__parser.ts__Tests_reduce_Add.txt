import { test, expect } from "vitest";

// ─── A reducer ─────────────────────────────────────────────────────────────

interface Todo {
  id: bigint;
  text: string;
  done: boolean;
  due: string | null;
}
interface State {
  todos: Todo[];
  filter: "all" | "open" | "done";
}
type Action =
  | { type: "add"; id: bigint; text: string }
  | { type: "toggle"; id: bigint }
  | { type: "filter"; filter: State["filter"] };

function reduce(state: State, action: Action): State {
  switch (action.type) {
    case "add":
      return {
        ...state,
        todos: [
          ...state.todos,
          { id: action.id, text: action.text, done: false, due: null },
        ],
      };
    case "toggle":
      return {
        ...state,
        todos: state.todos.map((t) =>
          t.id === action.id ? { ...t, done: !t.done } : t,
        ),
      };
    case "filter":
      return { ...state, filter: action.filter };
  }
}

test("Tests > reduce > Add", () => {
  const Empty: State = { todos: [], filter: "all" };
  const One = reduce(Empty, { type: "add", id: 1n, text: "buy milk" });
  const actual1 = One.todos;
  const expected1 = [{ id: 1n, text: "buy milk", done: false, due: null }];
  expect.soft(actual1).toEqual(expected1);
  const actual2 = One.todos[0]!.due;
  expect.soft(actual2).toEqual(null);
  const actual3 = One.todos[0]!.id;
  const expected3 = 1n;
  expect.soft(actual3).toEqual(expected3);
});
